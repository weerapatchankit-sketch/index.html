<script>
const $ = id => document.getElementById(id);
let timer;

/* ===== รายชื่อเกม: เพิ่มเกมใหม่แค่เติมบรรทัดในนี้ ===== */
const games = [
  {ic:"🖱️", name:"เกมกดปุ่มหนี", desc:"ปุ่มวิ่งหนีนิ้วคุณ", run:()=>startClick()},
  {ic:"❓", name:"เกมตอบคำถามกวนประสาท", desc:"ตัวเลือกสลับที่ตลอด", run:()=>startQuiz()},
];

/* พื้นหลังลอย */
["🤡","😈","😏","💀","🙄","🔥"].forEach((e,i)=>{
  for(let k=0;k<2;k++){
    const s=document.createElement("span");
    s.textContent=e;
    s.style.left=Math.random()*95+"%";
    s.style.animationDuration=(10+Math.random()*12)+"s";
    s.style.animationDelay=(-Math.random()*15)+"s";
    $("bg").appendChild(s);
  }
});

/* สร้างการ์ดเมนู */
games.forEach(g=>{
  const b=document.createElement("button");
  b.className="card";
  b.innerHTML='<span class="ic">'+g.ic+'</span><span><b>'+g.name+'</b><small>'+g.desc+'</small></span>';
  b.onclick=g.run;
  $("list").appendChild(b);
});

function show(html){
  $("menu").style.display="none";
  $("game").style.display="block";
  $("back").style.display="block";
  $("game").innerHTML=html;
}
function toMenu(){
  clearInterval(timer);
  $("game").style.display="none";
  $("back").style.display="none";
  $("menu").style.display="block";
}

/* ===== เกมกดปุ่มหนี ===== */
function startClick(){
  let n=0;
  const m=["ช้าไปนะ 😏","เกือบแล้ว~ (โกหก)","มือสั่นเหรอ?","ปุ่มไม่อยากให้กด","ลองอีกทีนะ 🤡"];
  show('<h2>🖱️ กดให้ครบ 10 ครั้ง</h2><div class="pill" id="t">เริ่มเลย!</div><button class="btn" id="b" style="position:fixed;left:35%;top:250px">กดเลย</button>');
  const b=$("b"), t=$("t");
  const mv=()=>{
    b.style.left=Math.random()*(innerWidth-110)+"px";
    b.style.top=200+Math.random()*(innerHeight-350)+"px";
  };
  b.ontouchstart=b.onmouseover=()=>{ if(Math.random()<0.7) mv(); };
  b.onclick=()=>{
    n++;
    t.innerText=m[Math.floor(Math.random()*m.length)]+" ("+n+"/10)";
    if(n>=10){ alert("ชนะแล้ว! ...ล้อเล่น เริ่มใหม่ 😈"); n=0; }
    mv();
  };
}

/* ===== เกมตอบคำถาม ===== */
function startQuiz(){
  const qs=[
    ["1 + 1 = ?",["2","11","หน้าต่าง"],0],
    ["ประเทศไทยมีกี่จังหวัด?",["77","76","เยอะมาก"],0],
    ["เมืองหลวงของญี่ปุ่นคือ?",["โตเกียว","โอซาก้า","ไม่บอก"],0],
    ["ข้อไหนคือคำตอบที่ถูก?",["ข้อนี้","ข้อนั้น","ไม่มีข้อไหนถูก"],2]
  ];
  const wrong=["ผิด! โง่จัง 🤡","มั่นใจมากนะ ผิดอ่ะ 😂","ใกล้เคียง (ไม่เลย)","สมองอยู่ไหนเนี่ย 🧠❓","ผิด แต่น่ารักดี 🙄"];
  const fake=["ถูก! ...แต่ขอถามใหม่อีกรอบ ไม่เชื่อ 😈","เกือบถูก ลองยืนยันอีกที!","ถูกนะ แต่แอบเปลี่ยนใจแล้ว กดใหม่"];
  let q=0, score=0, tricked=false, order=[0,1,2];
  function render(msg){
    const [text,opts]=qs[q];
    let h='<div class="pill">ข้อ '+(q+1)+'/'+qs.length+'</div><div class="pill">⭐ '+score+'</div><h2>'+text+'</h2><p style="min-height:24px">'+(msg||"")+'</p>';
    order.forEach(i=>{ h+='<button class="opt" onclick="ans('+i+')">'+opts[i]+'</button>'; });
    $("game").innerHTML=h;
  }
  window.ans=function(i){
    if(i===qs[q][2]){
      if(!tricked && Math.random()<0.6){ tricked=true; return render(fake[Math.floor(Math.random()*fake.length)]); }
      score++; q++; tricked=false;
      if(q>=qs.length){
        clearInterval(timer);
        $("game").innerHTML='<div class="logo">🏁</div><h2>จบเกม</h2><h3>คุณได้ '+score+'/'+qs.length+'</h3><p>'+(score==qs.length?"เก่งนะ... แต่เราจะบอกว่าได้ 0 เพราะเราไม่ชอบหน้าคุณ 😝":"ไปฝึกมาใหม่นะ 🤣")+'</p><button class="btn" onclick="startQuiz()">เล่นใหม่ (ถ้าไหว)</button>';
        return;
      }
      render("ถูก! ข้อต่อไป อย่าเพิ่งดีใจ 😏");
    } else {
      if(score>0 && Math.random()<0.3) score--;
      render(wrong[Math.floor(Math.random()*wrong.length)]);
    }
  };
  show("");
  render("ตอบให้ดีล่ะ ตัวเลือกจะสลับที่เรื่อย ๆ 😈");
  clearInterval(timer);
  timer=setInterval(()=>{ order.sort(()=>Math.random()-0.5); render(); },2500);
}

if("serviceWorker" in navigator) navigator.serviceWorker.register("sw.js");
</script>
